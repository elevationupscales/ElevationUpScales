async function uniqueCatalogSku(db,base){let sku=clean(base,100);for(let i=0;i<50;i+=1){const hit=await db.prepare("SELECT id FROM eus_inventory_items WHERE sku=? COLLATE NOCASE LIMIT 1").bind(sku).first();if(!hit)return sku;sku=clean(`${base}-${i+2}`,100);}throw new Error("Could not allocate a unique Catalog SKU");}
