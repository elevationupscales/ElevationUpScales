  const titleCounts=new Map();for(const r of rawRows){const key=normalizeTitle(r.title);if(key)titleCounts.set(key,(titleCounts.get(key)||0)+1);}for(const r of rawRows)r.variantGroup=(titleCounts.get(normalizeTitle(r.title))||0)>1;
  return{headers,rows:rawRows};
}
