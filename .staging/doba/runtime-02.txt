function exactSku(existing,itemNo,supplierSku){const current=clean(existing?.supplierSku,180).toLowerCase(),incoming=clean(supplierSku,180).toLowerCase(),combined=canonicalSku(itemNo,supplierSku).toLowerCase();return Boolean(incoming&&current&&(current===incoming||current===combined));}
function proposedChanges(existing,row,exact){const changes=[];if(!existing)return changes;if(!exact){changes.push("supplier SKU requires review");return changes;}if(row.derivedCostCents!==null&&existing.costCents!==row.derivedCostCents)changes.push(`supplier cost ${existing.costCents}→${row.derivedCostCents}`);if(existing.supplierStock!==row.inventoryQty)changes.push(`supplier stock ${existing.supplierStock??"unknown"}→${row.inventoryQty}`);if(row.sourceUrl&&row.sourceUrl!==existing.sourceUrl)changes.push("source URL refresh");if(row.inventoryQty===0&&existing.publishStatus==="published")changes.push("supplier unavailable — hold required");if(row.inventoryQty>=1&&row.inventoryQty<=10)changes.push("low supplier stock");return changes;}
function classification(existing,row,exact,variantGroup){if(!existing){if(row.inventoryQty===0)return{state:"NEW / HOLD — ZERO STOCK",blocker:"Supplier stock is 0; draft only."};if(variantGroup)return{state:"NEW / REVIEW — VARIANT GROUP",blocker:"Repeated source title indicates a variant group; review grouping before publish."};if(row.inventoryQty>=1&&row.inventoryQty<=10)return{state:"NEW / REVIEW — LOW STOCK",blocker:"Supplier stock is 1–10; conservative availability required."};return{state:"NEW / DRAFT",blocker:"New Doba Item No.; stage for merchandising/channel review."};}
  if(!exact)return{state:"REVIEW — SUPPLIER SKU CHANGED",blocker:"Item No. matches but exact supplier SKU does not. No silent substitution."};
  if(existing.publishStatus==="hold")return{state:"EXISTING HOLD — SOURCE REFRESH",blocker:existing.reviewState||"Existing Catalog HOLD remains in force."};
  if(row.inventoryQty===0)return{state:"UPDATE — SUPPLIER UNAVAILABLE",blocker:"Supplier stock is 0; product must not advertise as available."};
  if(row.inventoryQty>=1&&row.inventoryQty<=10)return{state:"UPDATE — LOW STOCK REVIEW",blocker:"Supplier stock is 1–10."};
  return{state:"UPDATE — EXACT MATCH",blocker:""};}

  const rawRows=rows.map((record,index)=>{const exportPriceCents=moneyCents(cellBy(record,map,"Dropshipping Price (US$)","Dropshipping Price"));const derivedCostCents=exportPriceCents===null?null:Math.round(exportPriceCents/(1+(markupPercent/100)));const inventoryQty=Math.max(0,int(cellBy(record,map,"Inventory Qty","Inventory Quantity"),0));const title=cellBy(record,map,"Product Name","Name","Title");const category=cellBy(record,map,"Category","Category Name","Product Category");const supplierSku=cellBy(record,map,"SKU Code","SKU","Supplier SKU");const itemNo=cellBy(record,map,"Item No.","Item No","Item Number");return{
    rowIndex:index+2,spuNo:cellBy(record,map,"SPU NO","SPU No.","SPU"),itemNo,supplierSku,title,brand:cellBy(record,map,"Brand","Brand Name"),category,sourceUrl:cellBy(record,map,"URL","Product URL"),

  };});
