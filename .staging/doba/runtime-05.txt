  await db.prepare(`INSERT INTO eus_catalog_meta(inventory_item_id,source_type,description,supplier_sku,supplier_stock,shipping_status,shipping_cents,primary_image,images_json,ebay_item_id,fourthwall_product_id,store_section,publish_status,review_state,created_by,updated_by,created_at,updated_at) VALUES(?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)`).bind(id,"doba",description,row.supplier_sku,qty,"unverified",null,images[0]||"",JSON.stringify(images),"","",storeSection(row.title,category),"draft",review,adminEmail,adminEmail,stamp,stamp).run();
  await db.prepare("UPDATE eus_doba_csv_rows SET catalog_product_id=?,catalog_sku=? WHERE id=?").bind(id,sku,row.id).run();row.catalog_product_id=id;row.catalog_sku=sku;await db.prepare(`INSERT INTO eus_catalog_events(id,inventory_item_id,sku,action,details_json,admin_email,created_at) VALUES(?,?,?,?,?,?,?)`).bind(uid("cat_evt"),id,sku,"doba_csv_draft_created",JSON.stringify({runId:run.id,itemNo:row.item_no,supplierSku:row.supplier_sku,derivedCostCents:row.derived_cost_cents,supplierStock:qty,prohibitedMarketplaces:safeJson(row.prohibited_marketplaces_json,[])}),adminEmail,stamp).run().catch(()=>{});return{id,sku};}
async function applyExisting(db,row,run,adminEmail){const product=await db.prepare(`SELECT i.id,i.sku,i.cost_cents,i.source_url,i.status,i.version,m.supplier_sku,m.supplier_stock,m.publish_status,m.review_state FROM eus_inventory_items i JOIN eus_catalog_meta m ON m.inventory_item_id=i.id WHERE i.id=? LIMIT 1`).bind(row.catalog_product_id).first();if(!product)return{review:true,reason:"Matched Catalog product no longer exists"};const exact=exactSku({supplierSku:product.supplier_sku},row.item_no,row.supplier_sku);if(!exact){const nextReview=clean(product.review_state,300)||"DOBA CSV: REVIEW — SUPPLIER SKU CHANGED";if(!clean(product.review_state,300))await db.prepare("UPDATE eus_catalog_meta SET review_state=?,updated_at=?,updated_by=? WHERE inventory_item_id=?").bind(nextReview,now(),adminEmail,product.id).run();await updateChannelState(db,{catalogProductId:product.id,sku:product.sku,stock:row.inventory_qty,status:"REVIEW REQUIRED",observed:"SUPPLIER SKU CHANGED",runId:run.id});return{review:true,reason:"Supplier SKU requires review"};}
  const stamp=now(),stock=int(row.inventory_qty,0),before={costCents:int(product.cost_cents,0),supplierStock:product.supplier_stock===null?null:int(product.supplier_stock,0),publishStatus:product.publish_status,reviewState:product.review_state,sourceUrl:product.source_url},changes=[];if(row.derived_cost_cents!==null&&int(product.cost_cents,0)!==row.derived_cost_cents)changes.push("cost");if((product.supplier_stock===null?null:int(product.supplier_stock,0))!==stock)changes.push("stock");if(row.source_url&&clean(product.source_url,1000)!==row.source_url)changes.push("source_url");
  await db.prepare("UPDATE eus_inventory_items SET supplier='doba',fulfillment_mode='dropship',cost_cents=?,source_url=?,version=version+1,updated_at=?,updated_by=? WHERE id=?").bind(row.derived_cost_cents??int(product.cost_cents,0),row.source_url||product.source_url,stamp,adminEmail,product.id).run();
  let publish=product.publish_status,review=product.review_state;if(stock===0&&publish==="published"){publish="hold";review="DOBA CSV: SUPPLIER OUT OF STOCK — REVIEW BEFORE REPUBLISH";changes.push("publish_hold");}
  await db.prepare("UPDATE eus_catalog_meta SET source_type='doba',supplier_sku=?,supplier_stock=?,publish_status=?,review_state=?,updated_at=?,updated_by=? WHERE inventory_item_id=?").bind(row.supplier_sku,stock,publish,review,stamp,adminEmail,product.id).run();
  await db.prepare(`INSERT INTO eus_catalog_events(id,inventory_item_id,sku,action,details_json,admin_email,created_at) VALUES(?,?,?,?,?,?,?)`).bind(uid("cat_evt"),product.id,product.sku,"doba_csv_reconciled",JSON.stringify({runId:run.id,itemNo:row.item_no,supplierSku:row.supplier_sku,before,after:{costCents:row.derived_cost_cents,supplierStock:stock,publishStatus:publish,reviewState:review},changes}),adminEmail,stamp).run().catch(()=>{});
  const syncStatus=stock===0||stock<=10?"REVIEW REQUIRED":"SYNCED",observed=stock===0?"OUT OF STOCK":stock<=10?"LOW STOCK":"SOURCE CURRENT";await updateChannelState(db,{catalogProductId:product.id,sku:product.sku,stock,status:syncStatus,observed,runId:run.id});return{updated:true,changes,publishStatus:publish};}
async function applyRun(db,runId,adminEmail){const run=await db.prepare("SELECT * FROM eus_doba_csv_runs WHERE id=? LIMIT 1").bind(runId).first();if(!run)throw new Error("Doba CSV run not found");if(run.status==="APPLIED"||run.status==="APPLIED WITH ERRORS")return runDetail(db,run.id);if(run.status!=="PREVIEW")throw new Error(`Run cannot be applied from status ${run.status}`);const rowsResult=await db.prepare("SELECT * FROM eus_doba_csv_rows WHERE run_id=? ORDER BY row_index").bind(run.id).all();const rows=rowsResult.results||[];let updated=0,staged=0,review=0,errors=0,holds=0;const itemNos=new Set();
  for(const row of rows){itemNos.add(clean(row.item_no,180).toLowerCase());if(row.applied_state==="APPLIED")continue;try{if(row.catalog_product_id){const result=await applyExisting(db,row,run,adminEmail);if(result.review)review+=1;else updated+=1;if(result.publishStatus==="hold")holds+=1;}else{await stageNewProduct(db,row,run,adminEmail);staged+=1;}await upsertSourceState(db,row,run);await db.prepare("UPDATE eus_doba_csv_rows SET applied_state='APPLIED',applied_at=? WHERE id=?").bind(now(),row.id).run();await saveEvent(db,{runId:run.id,rowId:row.id,catalogProductId:row.catalog_product_id,itemNo:row.item_no,supplierSku:row.supplier_sku,eventType:"row_applied",details:{classification:row.classification},adminEmail});}catch(error){errors+=1;await db.prepare("UPDATE eus_doba_csv_rows SET applied_state='ERROR',blocker=? WHERE id=?").bind(clean(error?.message,500),row.id).run();await saveEvent(db,{runId:run.id,rowId:row.id,catalogProductId:row.catalog_product_id,itemNo:row.item_no,supplierSku:row.supplier_sku,eventType:"row_apply_error",status:"ERROR",details:{message:clean(error?.message,500)},adminEmail});}}
