function parseCsv(text){
  const input=String(text??"").replace(/^\uFEFF/,"");
  if(!input.trim())throw new Error("CSV file is empty");
  const rows=[];let row=[],cell="",quoted=false;
  for(let i=0;i<input.length;i+=1){const ch=input[i];if(quoted){if(ch==='"'){if(input[i+1]==='"'){cell+='"';i+=1;}else quoted=false;}else cell+=ch;continue;}if(ch==='"'){quoted=true;continue;}if(ch===','){row.push(cell);cell="";continue;}if(ch==='\n'||ch==='\r'){if(ch==='\r'&&input[i+1]==='\n')i+=1;row.push(cell);cell="";if(row.some(v=>String(v).trim()!==""))rows.push(row);row=[];continue;}cell+=ch;}
  row.push(cell);if(row.some(v=>String(v).trim()!==""))rows.push(row);
  if(rows.length<2)throw new Error("CSV must contain a header row and at least one product row");
  const headers=rows.shift().map(v=>clean(v,240));
  return{headers,rows};
}
